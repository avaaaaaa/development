* when using promises directly, avoid try-catch
when using promises via async-await, we can use try-catch or .catch(callback)


const getNotifications = async () => {
	const url = "";
	const response = await fetch(url);
	const data = await response.json();
	return data;
}


// handle errors

getNotifications()
	.then(data => {
		console.log(data);
	}).catch(error => {
		console.log(error.message);
	});


// handle errors

const init = async () => {
	try{
		const data = await getNotifications();
		console.log(data);
	} catch(error){
		console.error(error.message);
	}
}

init();


// below does not work
// we should directly put the function that might throw an error within try-catch
// in this case, init() does not throw an error, but another function within init() throws an error
// but that function (getNotifications()) is not wrapped with try-catch
// thus the error could not be silenced

const init = async () => {
	const data = await getNotifications();
	console.log(data);
}

try{
	init();
} catch(error){

}


* return inside an async function will resolve the promise successfully (even if the return was inside a catch { ... } block).


const getUnreadCount = async () => {
    try {
        const response = await fetch("https://jsdemo-3f387-default-rtdb.europe-west1.firebasedatabase.app/notifications/new.json");
        const data = await response.json();
        return data.count;
    } catch (error) {
        // ❌ this will give you an unexpected result
        return "An error has occurred";
    }
}

getUnreadCount().then(data => {
    console.log(data, "+");
}).catch(error => {
    console.error(error, "-");
});


output is normally "3 +"
now, remove a character from the url
output is now "An error has occured +", so the return statement resolved the promise
what we would expect is "An error has occured -", because the promise should have been rejected 

what happens if we put  throw "An error has occured"  instead of  return "An error has occured"
output is now "An error has occured -", as expected


* example

const checkForNewNotifications = async () => {
    try{
        const response = await fetch("https://jsdemo-3f387-default-rtdb.europe-west1.firebasedatabase.app/notifications/new.json");
        const data = await response.json();
        console.log(data);
        showNewNotifications(data.count);
    
    } catch(error){
        showNewNotifications();
    }
    
}

function showNewNotifications(count) {
    if (!count) {
        console.log("We could not load your notifications. Try again later.");
        return;
    }
    console.log(`You have ${count} new notifications. Would you like to read them?`);
}

// sample usage
checkForNewNotifications();


* finally block
The finally { ... } block will run after either the try { ... } or catch { ... } block.
it can help you avoid some code duplication similar to .finally(callback)


const init = async () => {
    try {
        await wait(1_000);
        //return 42;
    } catch (error) {
        console.log("An error has occurred");
        console.error(error);
        //return 42;
    } finally {
    	return 42;
    }
}

// instead of repeating  return 42;  within try and catch, we can put it within finally
