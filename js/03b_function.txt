* improving the algorithm
function isPositive(n){
  if(n > 0){
    return true
  }
  else{
    return false
  }
}
// can also be written as below
function isPositive(n){
  return n > 0
}

function getMessage(n){
  let msg
  if(n > 0){
    msg = "positive"
  }
  else{
    msg = "not positive"
  }
  return msg
}
// can also be written as below
function getMessage(n){
  let msg = "not positive"
  if(n > 0){
    msg = "positive"
  }
  return msg
}

* if program flow reaches to the end of execution of a function and haven't encountered a 'return' statement until that point, js will implicitly add 'return undefined" to the end of code block where
program flow had reached

function f(n){
  if(n > 5){
    return "> 5"
  } else if(n < 5){
    return "< 5"
  }
  // f(5) reaches and stops at this line but there is no 'return' statement there
  // so, js imlicitly add 'return undefined' here
}

console.log(f(5))
// undefined


* 'return' and program flow
if a function reaches a return statement, the function will be popped off from the call stack
and the function at top of the call stack will continue from where it called the previous function

function f1(){
  f2()
  console.log("f1")
}

function f2(){
  console.log("f2")
}

f1()
// "f2"
// "f1"
// if 'return' was added before f2() call,
// the console log output would only show "f2" 


* lines of code after 'return' statement are not accessible, i.e. won't be executed
function f(){
  let x = 1
  return x
  x = 2
  return x
}
console.log(f())
// 1
