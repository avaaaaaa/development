* function as a parameter
functions can be passed as a parameter

function f1(parFunc){
	if(true){
		parFunc();
	}
}

function f2(){
    console.log("hello world");
}

f1(f2);
// hello world


* variable scope
variables that are defined within a code block can only be accessed within that code block
variables that are defined outside a code block can be accessed within that code block

function f1(){
  let v1 = 1
}

function f2(){
  console.log(v1)
}

f2()
// ReferenceError: v1 is not defined


const a = 3;

function test(){
  console.log(a);
}

test();
// 3


let a = 9;

function test(){
  a = 20;
}

test();
console.log(a);
// 20

* improving the algorithm
function isPositive(n){
  if(n > 0){
    return true
  }
  else{
    return false
  }
}
// can also be written as below
function isPositive(n){
  return n > 0
}

function getMessage(n){
  let msg
  if(n > 0){
    msg = "positive"
  }
  else{
    msg = "not positive"
  }
  return msg
}
// can also be written as below
function getMessage(n){
  let msg = "not positive"
  if(n > 0){
    msg = "positive"
  }
  return msg
}
// or use ternary operator: return (n > 0) ? "positive" : "not positive"

* if the program flow reaches to the end of execution of a function and haven't encountered a 'return' statement until that point, js will implicitly add 'return undefined" to the end of code block where
the program flow had reached

function f(n){
  if(n > 5){
    return "> 5"
  } else if(n < 5){
    return "< 5"
  }
  // f(5) reaches and stops at this line but there is no 'return' statement there
  // so, js implicitly add 'return undefined' here
}

console.log(f(5))
// undefined


* 'return' and program flow
if a function reaches a return statement, the function will be popped off from the call stack
and the function at top of the call stack will continue from where it called the previous function

function f1(){
  f2()
  console.log("f1")
}

function f2(){
  console.log("f2")
}

f1()
// "f2"
// "f1"
// if 'return' was added before f2() call,
// only "f2" would be on the console output 


* lines of code after 'return' statement are not accessible, i.e. won't be executed
function f(){
  let x = 1
  return x
  x = 2
  return x
}
console.log(f())
// 1
