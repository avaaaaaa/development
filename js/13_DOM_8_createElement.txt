* document.createElement()
this approach is often used when dynamically injecting an element, a script or a stylesheet
there is no difference in terms of performance between createElement() and insertAdjacentHTML()


const htmlString = `<p class="text-center">Hello World</p>`;
document.body.insertAdjacentHTML("beforeend", htmlString);

// the same as:

const paragraph = document.createElement("p");
paragraph.classList.add("text-center");
paragraph.textContent = "Hello World";
document.body.append(paragraph);
	We can also do the following:
	document.body.appendChild(paragraph);
	document.body.insertAdjacentElement("beforeend", paragraph);


// assuming there is a function called isDarkMode()
let mode = "light-mode";
if(isDarkMode()){
	mode = "dark-mode";
}
const htmlString = `<div class="${mode}">Hello World</div>`;
document.body.insertAdjacentHTML("afterbegin", htmlString);

// the same as:

const div = document.createElement("div");
div.textContent = "Hello World";

if(isDarkMode()){
	div.classList.add("dark-mode");
} else{
	div.classList.add("light-mode");
}

document.body.prepend(div);
	the following can also be used:
	document.body.insertAdjacentElement("afterbegin", div)
