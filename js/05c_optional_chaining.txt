* accessing properties from the properties that yield either null or undefined will cause TypeError
to avoid getting this error, we can use optional chaining with ?. operator
with that, it will return undefined rather than error
with this feature, we don't need an if-condition to avoid errors


* ideally, TypeScript can be used to handle these issues and enforce types in your code base.


const user = {
  details: {
    name: "gg",
    age: 20
  },
  data: null
};

console.log(user.details.location.address);  // TypeError, user.details.location is undefined
console.log(user.details.location?.address);  // undefined

console.log(user.data.id);  // TypeError, user.data is null
console.log(user.data?.id);  // undefined


*
let name = undefined;
if (user.details && user.details.name && user.details.name.firstName) {
    name = user.details.name.firstName;
}
// with optional chaining, we can refactore this code above into the following
// const name = user.details?.name?.firstName;
// but user can be null or undefined too, so it is better to add ?. after user
// const name = user?.details?.name?.firstName;
// this can be interpreted as
//		is there "user" object, because we need to access "details" from "user"
//		is there "details" property, because we need to access "name" from "details"
//		is there "name" property, because we need to access "firstName" from "name"


* array

const data = {
    temperatures: [-3, 14, 4]
}

let firstValue = undefined;
if (data.temperatures) {
    firstValue = data.temperatures[0];
}
// const firstValue = data.temperatures?.[0];
// with that, we avoid null[0] or undefined[0] which will cause TypeError


* method call

const upperCasedName = person.name?.toUpperCase();
// either null.toUpperCase() or undefined.toUpperCase() will cause TypeError
// with optional chaining, the code above will return undefined

const upperCasedName = person.name?.toUpperCase?.();
// if person.name is a number, calling toUpperCase() will cause TypeError
// to avoid that, we can use optional chaining to call the method toUpperCase


* update values of existed keys only
// see * using && as an if-condition in 02a_if_condition.txt
const settings = {
     theme: "light"
};

settings?.theme && (settings.theme = "dark");
console.log(settings); // {theme: "dark"}
// if we set "theme" as nullish value in the definition, this won't work

// we can do the following if "theme" is set as nullish value
"theme" in settings && (settings.theme = "dark");
console.log(settings); // {theme: "dark"}
