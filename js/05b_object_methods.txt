* Object.keys(<object>)
returns an array of keys of <object>


* Object.values(<object>)
returns an array of values of <object>


const person = {
  id : 1,
  name: "gg",
  interests : ["a", "b"],
}

Object.keys(person);
// [ 'id', 'name', 'interests' ]
Object.values(person);
// [ 1, 'gg', [ 'a', 'b' ] ]


* Object.entries(<object>)
returns an array that pairs keys and corresponding values within separate arrays


const person = {
  id : 1,
  name: "gg",
  interests : ["a", "b"],
}

const entries = Object.entries(person);
// [["id", 1], ["name", "gg"], ["interests", ["a", "b"]]]
// entries[0] is ["id", 1]
// entries[1] is ["name", "gg"]
// entries[2] is ["interests", ["a", "b"]]


* Object.groupBy(<array of objects>, <callback func>)
const books = [
  {id: 1,
   author: "gg",
   pages: 200,},
  {id: 2,
  author: "ff",
  pages: 300},
  {id: 3,
  author: "gg",
  pages: 250},
];

const groupedByAuthor = Object.groupBy(books, book => book.author);
/*
{
  gg: [
    { id: 1, author: 'gg', pages: 200 },
    { id: 3, author: 'gg', pages: 250 }
  ],
  ff: [ { id: 2, author: 'ff', pages: 300 } ]
}
*/
