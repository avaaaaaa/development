* string to number
// when reading values from textboxes, they are stored as strings even if they have only numbers
// so, we have to convert this string to the number

let string = "35";
let radix = 10;
let number = Number.parseInt(string, radix);
console.log(string, radix, number);
// '35' 10 35
// radix parameter is optional and it doesn't always default to 10
// thus, it is recommended to specifiy explicitly
// there could be some edge cases where skipping radix might cause unexpected outcomes

Number.parseInt("123abc", 10); // 123
Number.parseInt("5 meters", 10); // 5
Number.parseInt("a50", 10); // NaN
