we can use html entities in js to put symbols on web pages
web browsers will render these entities and put the symbol afterwards
this is different from putting symbols on the console with unicode characters
we are trying to put these symbols on actual web page

they start with &(ampersand) and ends with semi-colon(;)
&copy; for copyright symbol (©)
&eur; for euro symbol (€)

let x = "&copy;";
console.log(x);
// '&copy;'
// as explained earlier, html entities are used to put symbols on web pages i.e. html files
// so, 'x' won't be shown as copyright symbol on the console
// to do that, we can use unicode equivalent -> let x = "\u{00A9}";  // '©'

---weather.js---
export default class Weather {
    /** @param {object} details */
    constructor(details){
        this.details = details;
    }
    getLocation(){
        let city = this.details.location.name;
        let country = this.details.location.country;
        return city + ", " + country;
    }
    getIcon(){
        return this.details.current.weather_icons[0];
    }
    getTemperature(){
        return this.details.current.temperature + "&deg;" + "C";
    }
}


---index.js---
import data from "./data.js";
import Weather from "./weather.js";

let cityCountry = document.querySelector("#city-country");
let icon = document.querySelector("#icon");
let temperature = document.querySelector("#temperature");

// Instantiate Weather with `data`. This data becomes the `details` parameter in the Weather class
let weather = new Weather(data);

cityCountry.textContent = weather.getLocation();
icon.src = weather.getIcon();
temperature.innerHTML = weather.getTemperature();
